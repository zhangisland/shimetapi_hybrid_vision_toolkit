cmake_minimum_required(VERSION 3.16)
project(aps_host_tests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
enable_testing()
set(APS_OPENCV_ROOT "" CACHE PATH "Optional existing Windows OpenCV DLL distribution")
if(WIN32 AND APS_OPENCV_ROOT)
 set(OpenCV_INCLUDE_DIRS "${APS_OPENCV_ROOT}/include")
 foreach(component core imgproc imgcodecs highgui)
  list(APPEND OpenCV_LIBS "${APS_OPENCV_ROOT}/lib/opencv_${component}4.lib")
 endforeach()
else()
 find_package(OpenCV REQUIRED COMPONENTS core imgproc imgcodecs highgui)
endif()
add_executable(aps_tests test_aps.cpp)
target_include_directories(aps_tests PRIVATE ../include ${OpenCV_INCLUDE_DIRS})
target_link_libraries(aps_tests PRIVATE ${OpenCV_LIBS})
if(MSVC)
 target_compile_options(aps_tests PRIVATE /utf-8)
endif()
add_test(NAME aps_pipeline COMMAND aps_tests)

add_executable(aps_live_fake ../samples/cpp/live_record_display/main.cpp ../samples/cpp/live_record_display/live_widgets.cpp fake_sdk.cpp)
add_executable(aps_record_fake test_record.cpp ../samples/cpp/live_record_display/live_widgets.cpp fake_sdk.cpp)
add_library(aps_player_compile OBJECT ../samples/cpp/player/player_widgets.cpp)
foreach(target aps_live_fake aps_record_fake aps_player_compile)
 target_include_directories(${target} PRIVATE ../include ${OpenCV_INCLUDE_DIRS})
 target_link_libraries(${target} PRIVATE ${OpenCV_LIBS})
 if(MSVC)
  target_compile_options(${target} PRIVATE /utf-8)
 endif()
endforeach()
add_test(NAME aps_live_fake COMMAND aps_live_fake --no-display --aps-bayer gbrg --aps-wb once)
add_test(NAME aps_hardware_rejected COMMAND aps_live_fake --aps-exposure-us 5000)
set_tests_properties(aps_hardware_rejected PROPERTIES WILL_FAIL TRUE)
add_test(NAME aps_record_fake COMMAND aps_record_fake)

add_executable(x5_controls_contract test_x5_controls.cpp)
target_include_directories(x5_controls_contract PRIVATE ../include)
add_test(NAME x5_controls_contract COMMAND x5_controls_contract)
add_library(hvs_recorder_compile OBJECT ../samples/cpp/hvs_record/main.cpp)
target_include_directories(hvs_recorder_compile PRIVATE ../include)

add_executable(hvs_storage_test test_recording_storage.cpp)
add_test(NAME hvs_storage_test COMMAND hvs_storage_test)
if(MSVC)
 target_compile_options(hvs_recorder_compile PRIVATE /utf-8)
endif()

add_executable(aps_metadata_test test_aps_metadata.cpp)
target_include_directories(aps_metadata_test PRIVATE ../include)

if(MSVC)
 target_compile_options(aps_metadata_test PRIVATE /utf-8)
endif()
